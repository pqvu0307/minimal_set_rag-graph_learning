\section{bài toán}

\begin{frame}[fragile]{rag là gì, và nó thiếu gì}
  \lead{hỏi một model ngôn ngữ về sự kiện có thật thì nó hay \emph{bịa}.
        cách chữa phổ biến tên là \hl{rag}: trước khi trả lời, đi tìm vài
        đoạn văn bản liên quan rồi đưa kèm câu hỏi cho model đọc.}

  \begin{dia}
câu hỏi  ->  [ĐI TÌM]  ->  40 đoạn văn bản  ->  [MODEL ĐỌC]  ->  câu trả lời
                                ^
                                |
                       trong 40 đoạn này, đoạn nào THẬT SỰ tạo ra
                       câu trả lời?  hiện không ai biết.
  \end{dia}

  \vspace{0.4em}
  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{badb}[title={vì sao «không ai biết» là vấn đề}]
      câu trả lời không kèm nguồn là câu trả lời \no{không dùng được} trong y tế,

    \end{badb}
  \end{column}
  \begin{column}{0.49\textwidth}
    \begin{bx}[title={và nó cũng chặn việc cải thiện hệ thống}]
      không biết đoạn nào đúng thì \no{không dạy} được bộ đi tìm lần sau

    \end{bx}
  \end{column}

\begin{frame}[fragile]{ví dụ chạy suốt buổi}
  \framesubtitle{mọi bước sau đều quay lại ví dụ này}
\begin{dia}
câu hỏi   ai đạo diễn bộ phim thắng giải oscar phim hay nhất năm 1994?
đáp án    Robert Zemeckis

hệ thống lấy về 40 đoạn.  ba đoạn đáng chú ý:

  đoạn  3   "Forrest Gump thắng oscar phim hay nhất năm 1994."
  đoạn 12   "Robert Zemeckis đạo diễn Forrest Gump."
  đoạn 19   "Forrest Gump là hiện tượng văn hóa của thập niên 1990."

  ... và 37 đoạn khác, phần lớn không liên quan
\end{dia}

  \vspace{0.3em}
  \begin{columns}[t]
  \begin{column}{0.47\textwidth}
    \begin{keyb}[title={đáp số mà tầng a phải tự tìm ra}]
      \hl{đúng hai đoạn: 3 và 12.}
      \vspace{0.25em}

      đoạn 3 cho biết \emph{phim nào}, đoạn 12 cho biết \emph{ai đạo diễn}.
      \no{không đoạn nào một mình} trả lời được — phải \hl{bắc cầu} qua hai đoạn.
    \end{keyb}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{badb}[title={đoạn 19 là cái bẫy của cả bài}]
      đoạn 19 \no{không} nói ai đạo diễn, cũng \no{không} nói phim thắng giải.
      \vspace{0.25em}

      nhưng nó \emph{nhắc tên phim} — và chỉ cần thế là model nhớ ra đáp án từ
      kiến thức sẵn có rồi trả lời đúng. ta sẽ phải loại nó.
    \end{badb}
  \end{column}
  \end{columns}

  \end{column}
  \begin{column}{0.48\textwidth}

      \vspace{0.3em}

      \dd{ví dụ: câu hỏi $\to$ «Forrest Gump» $\to$ «Zemeckis» $\to$ đoạn 12}

  \end{column}
  \end{columns}

  \vspace{0.7em}
  \begin{keyb}
    hai tầng nối nhau bằng \hl{một thứ duy nhất}: tầng a trả về tập đoạn $T$,
    tầng b nhận $T$ rồi đi tìm đường. \no{không điểm số nào đi qua ranh giới} —
    lý do ở cuối buổi.
  \end{keyb}

\section{tầng a}

\begin{frame}[fragile]{40 đoạn ứng viên ở đâu ra}
  \framesubtitle{trước c0}
  \lead{trước khi làm gì khác, phải có một rổ đoạn để chọn. rổ này gọi là
        \hl{pool}, ký hiệu $C(q)$ — «$q$» là câu hỏi.}

  \begin{columns}[t]
  \begin{column}{0.52\textwidth}
    \begin{bx}[title={ba nguồn, mỗi nguồn một hạn ngạch}]
      \begin{dia}
bm25        20 đoạn   khớp TỪ KHOÁ
gfm-rag     15 đoạn   đi theo ĐỒ THỊ
ngẫu nhiên   5 đoạn   lấy bừa trong kho
                    ----
                      40 đoạn = pool C(q)
      \end{dia}
      \vspace{0.25em}
      \dd{thô tối đa 60, khử trùng lặp, rồi xếp hạng lại \emph{trong từng nguồn}
          — không trộn chung, không thì nguồn mạnh nuốt hết chỗ}
    \end{bx}
  \end{column}
  \begin{column}{0.44\textwidth}
    \begin{bx}[title={vì sao phải có nguồn ngẫu nhiên}]
      nếu pool chỉ gồm đoạn mà bộ đi tìm \emph{cho là} đúng, thì nhãn ta đào ra
      chỉ xác nhận lại thành kiến của chính nó.
      \vspace{0.3em}

      5 đoạn bừa là \hl{đối chứng}: nếu chúng cũng hay lọt vào đáp số thì phép đo
      của ta đang hỏng.
    \end{bx}
  \end{column}
  \end{columns}

  \vspace{0.4em}
  \begin{keyb}
    cả bài toán từ đây là: trong $2^{40}$ tập con của 40 đoạn này,
    \hl{chỉ ra đúng một tập}. không thể thử hết — $2^{40}$ là hơn nghìn tỉ.
  \end{keyb}
\end{frame}

\begin{frame}[fragile]{c0 · chẻ câu hỏi thành mảnh}

  \lead{chẻ \hl{câu hỏi \emph{và} đáp án vàng} thành những mẩu sự thật nhỏ — ta

  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{bx}[title={vào là $q$ \emph{và} $y$ — ra là $J$ mệnh đề}]
      \begin{dia}
vào   q  ai đạo diễn phim thắng oscar 1994?
      y  Robert Zemeckis

ra    J = 2 mảnh
  mảnh 1  phim thắng oscar 1994 là

  mảnh 2  Forrest Gump do Zemeckis

      \end{dia}
    \end{bx}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{badb}[title={chỗ dễ hỏng: thực thể bắc cầu}]
      \no{«Forrest Gump» không nằm trong $q$ lẫn $y$.} bắt llm \emph{nhớ} thì

      \no{thoát ở c0 như thể pool thiếu}.
    \end{badb}
    \begin{keyb}[title={chữa: cho llm \emph{đọc} thay vì \emph{nhớ}}]
      «Forrest Gump» nằm sẵn trong \hl{đoạn 3 của pool} — nên đưa pool vào prompt,
      và bắt mỗi mảnh kèm \hl{trích nguyên văn}.
      \vspace{0.25em}

      \dd{trích không có trong pool = llm đang bịa, loại bằng phép so chuỗi.
          vẫn \textbf{1 lượt mỗi truy vấn}}
    \end{keyb}
  \end{column}
  \end{columns}

\begin{frame}[fragile]{c0 · đối chiếu từng đoạn với từng mảnh}
  \framesubtitle{c0}
  \lead{với mỗi cặp (đoạn $k$, mảnh $j$), hỏi một model nhỏ \hl{chạy trên máy mình}:
        \emph{«đọc đoạn này có suy ra được câu kia không?»}}

  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{keyb}[title={công cụ: một model nli nhỏ}]
      \begin{dia}
premise     nội dung đoạn k     ~100 chữ
hypothesis  mảnh j              1 câu
                 |
                 v
         entail / neutral / contradict
      \end{dia}
      \vspace{0.25em}
      \dd{model ${\sim}400$M tham số, \yes{0 lượt gọi api}, $40 \times J$ lượt
          chạy — vài giây trên máy mình}
    \end{keyb}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{badb}[title={nên mảnh PHẢI là câu khẳng định}]
      nli chỉ được huấn luyện trên mệnh đề. đưa thẳng
      \texttt{"Who directed X? -> Y"} vào ô \emph{hypothesis} là ngoài phân phối —
      \no{bảng phủ thành rác, mà không một cảnh báo nào}.
      \vspace{0.25em}

      đây là lý do bước viết lại ở slide trước \hl{không bỏ qua được}.
    \end{badb}
    \begin{bx}[title={kết quả gọi là bảng phủ}]
      ô $(k, j)$ đánh dấu nếu đoạn $k$ \emph{nói ra} mảnh $j$ — slide sau là bảng
      thật của ví dụ.
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{vì sao không dừng ngay ở đây?}
  \framesubtitle{c0}
  \lead{trích dẫn đã chỉ thẳng mảnh nào ← đoạn nào. hợp lại là một tập ứng viên,
        \hl{miễn phí}. vậy sao không trả luôn nó rồi xong?}

  \begin{columns}[t]
  \begin{column}{0.46\textwidth}
    \begin{keyb}[title={và ta có dùng nó thật}]
      \begin{dia}
mảnh 1 <- trích trong đoạn  3
mảnh 2 <- trích trong đoạn 12
                 |
                 v
        T_llm = {3, 12}     0 lượt thêm
      \end{dia}
      \vspace{0.25em}
      đây thành \hl{ứng viên đầu tiên} mà \texttt{c5} thử — trước cả danh sách
      nghiệm phủ. đúng ngay thì phần tìm kiếm \yes{bỏ qua được}.
    \end{keyb}
  \end{column}
  \begin{column}{0.50\textwidth}
    \begin{badb}[title={nhưng nó là một \emph{lời khai}, không phải một \emph{phép đo}}]
      \textbf{1}\; chưa ai kiểm. cả thiết kế này đứng trên chỗ đó: $v(S)$ là
      \emph{chạy thật} model trên $S$ rồi đối đáp án. llm tự khai nó dùng đoạn nào
      là đúng loại việc \no{model vốn làm kém} — $35$–$62\%$ ca nó trả lời đúng
      \emph{khi ngữ cảnh không đủ}.
      \vspace{0.3em}

      \textbf{2}\; mỗi mảnh nó chỉ trích \emph{một} đoạn. mà mảnh 1 còn được đoạn 7
      đỡ nữa — nên ta \no{mất cả họ} $W$, mất luôn độ tin cậy và chỉ số cấu trúc.
      \vspace{0.3em}

      \textbf{3}\; không có gì nói tập con của nó không cũng đủ.
    \end{badb}
  \end{column}
  \end{columns}

  \vspace{0.3em}
  \begin{keyb}

          trung thực; nhưng dùng lúc đào nhãn thì sai — nó là thứ model
