\section{tầng a}

\begin{frame}[fragile]{40 đoạn ứng viên ở đâu ra}
  \framesubtitle{trước c0}
  \lead{trước khi làm gì khác, phải có một rổ đoạn để chọn. rổ này gọi là
        \hl{pool}, ký hiệu $C(q)$ — «$q$» là câu hỏi.}

  \begin{columns}[t]
  \begin{column}{0.52\textwidth}
    \begin{bx}[title={ba nguồn, mỗi nguồn một hạn ngạch}]
      \begin{dia}
bm25        20 đoạn   khớp TỪ KHOÁ
gfm-rag     15 đoạn   đi theo ĐỒ THỊ
ngẫu nhiên   5 đoạn   lấy bừa trong kho
                    ----
                      40 đoạn = pool C(q)
      \end{dia}
      \vspace{0.25em}
      \dd{thô tối đa 60, khử trùng lặp, rồi xếp hạng lại \emph{trong từng nguồn}
          — không trộn chung, không thì nguồn mạnh nuốt hết chỗ}
    \end{bx}
  \end{column}
  \begin{column}{0.44\textwidth}
    \begin{bx}[title={vì sao phải có nguồn ngẫu nhiên}]
      nếu pool chỉ gồm đoạn mà bộ đi tìm \emph{cho là} đúng, thì nhãn ta đào ra
      chỉ xác nhận lại thành kiến của chính nó.
      \vspace{0.3em}

      5 đoạn bừa là \hl{đối chứng}: nếu chúng cũng hay lọt vào đáp số thì phép đo
      của ta đang hỏng.
    \end{bx}
  \end{column}
  \end{columns}

  \vspace{0.4em}
  \begin{keyb}
    cả bài toán từ đây là: trong $2^{40}$ tập con của 40 đoạn này,
    \hl{chỉ ra đúng một tập}. không thể thử hết — $2^{40}$ là hơn nghìn tỉ.
  \end{keyb}
\end{frame}

\begin{frame}[fragile]{c0 · chẻ câu hỏi thành mảnh}
  \framesubtitle{c0}
  \lead{chẻ \hl{câu hỏi \emph{và} đáp án vàng} thành những mẩu sự thật nhỏ cần
        phải có — ta gọi là \hl{mảnh}. tầng a chạy \emph{offline để đào nhãn},
        nên có $y$ trong tay.}

  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{bx}[title={vào là $q$ \emph{và} $y$ — ra là $J$ mệnh đề}]
      \begin{dia}
vào   q  ai đạo diễn phim thắng oscar 1994?
      y  Robert Zemeckis

ra    J = 2 mảnh
  mảnh 1  phim thắng oscar 1994 là
          Forrest Gump
  mảnh 2  Forrest Gump do Zemeckis
          đạo diễn
      \end{dia}
    \end{bx}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{bx}[title={lấy mảnh ở đâu}]
      \hl{musique} cho sẵn \emph{(câu hỏi con, đáp án con)} — nhưng đó chưa phải
      mệnh đề, phải viết lại.
      \vspace{0.25em}

      \hl{corpus khác} thì nhờ llm, \textbf{1 lượt cho cả truy vấn} — không phải
      mỗi tập con.
    \end{bx}
    \begin{badb}[title={được đọc $y$, nhưng vẫn còn một lỗ}]
      \no{«Forrest Gump» không nằm trong $q$ lẫn $y$} — nó là thực thể bắc cầu.
      musique cho sẵn; corpus khác thì llm phải tự biết, mà đoán sai thì truy vấn
      \no{thoát ở c0 như thể pool thiếu}.
    \end{badb}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{c0 · đối chiếu từng đoạn với từng mảnh}
  \framesubtitle{c0}
  \lead{với mỗi cặp (đoạn $k$, mảnh $j$), hỏi một model nhỏ \hl{chạy trên máy mình}:
        \emph{«đọc đoạn này có suy ra được câu kia không?»}}

  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{keyb}[title={công cụ: một model nli nhỏ}]
      \begin{dia}
premise     nội dung đoạn k     ~100 chữ
hypothesis  mảnh j              1 câu
                 |
                 v
         entail / neutral / contradict
      \end{dia}
      \vspace{0.25em}
      \dd{model ${\sim}400$M tham số, \yes{0 lượt gọi api}, $40 \times J$ lượt
          chạy — vài giây trên máy mình}
    \end{keyb}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{badb}[title={nên mảnh PHẢI là câu khẳng định}]
      nli chỉ được huấn luyện trên mệnh đề. đưa thẳng
      \texttt{"Who directed X? -> Y"} vào ô \emph{hypothesis} là ngoài phân phối —
      \no{bảng phủ thành rác, mà không một cảnh báo nào}.
      \vspace{0.25em}

      đây là lý do bước viết lại ở slide trước \hl{không bỏ qua được}.
    \end{badb}
    \begin{bx}[title={kết quả gọi là bảng phủ}]
      ô $(k, j)$ đánh dấu nếu đoạn $k$ \emph{nói ra} mảnh $j$ — slide sau là bảng
      thật của ví dụ.
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{c0 · bảng phủ, và nó bắt ngay cái bẫy}
  \framesubtitle{c0}
  \begin{columns}[t]
  \begin{column}{0.5\textwidth}
    \begin{bx}[title={bảng phủ của ví dụ — 0 lượt llm}]
      \begin{dia}
            mảnh 1   mảnh 2
đoạn  3     CÓ          —
đoạn  7     CÓ          —
đoạn 12       —        CÓ
đoạn 23       —        CÓ
đoạn 31       —        CÓ
đoạn 19       —         —     <- cái bẫy
... 34 đoạn còn lại đều trống
      \end{dia}
      \vspace{0.25em}
      \hl{đoạn 19 trống cả hai cột} — nó không nói ra mẩu sự thật nào. bảng phủ
      loại nó mà \yes{không tốn một lượt llm nào}.
    \end{bx}
  \end{column}
  \begin{column}{0.46\textwidth}
    \begin{bx}[title={viết lại cho gọn}]
      với một tập đoạn $S$, gọi $\mathrm{cov}(S)$ là tập các mảnh mà $S$ nói ra được:
      \begin{align*}
        \mathrm{cov}(S) &= \{\, j : \exists k \in S,\; (k,j) = \text{CÓ} \,\}\\[0.15em]
        \mathrm{adq}(S) &\iff \mathrm{cov}(S) = \{1, \dots, J\}
      \end{align*}
      \dd{$\mathrm{adq}$ đọc là «\emph{adequate}» — $S$ \textbf{phủ hết} mọi mảnh}
      \vspace{0.3em}

      \dd{ví dụ: $\mathrm{cov}(\{3\}) = \{1\}$ nên $\mathrm{adq}(\{3\})$ sai ·
          $\mathrm{cov}(\{3,12\}) = \{1,2\}$ nên $\mathrm{adq}(\{3,12\})$ \yes{đúng}}
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{c0 · vì sao phải chẻ ra, thay vì hỏi thẳng}
  \framesubtitle{c0}
  \lead{câu hỏi tự nhiên: sao không đưa thẳng cả tập $S$ và cả câu hỏi cho model
        nli? hai lý do, độc lập nhau.}

  \begin{columns}[t]
  \begin{column}{0.48\textwidth}
    \begin{bx}[title={lý do 1 · độ dài}]
      cả tập $S$ có thể dài tới $10{,}000$ chữ, mà model nli chỉ đọc được $512$.
      \vspace{0.4em}

      chẻ ra thì mỗi lần kiểm chỉ là \textbf{1 đoạn} (${\sim}100$ chữ) đối
      \textbf{1 câu ngắn}. vừa khít.
    \end{bx}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{badb}[title={lý do 2 · bắc cầu}]
      hỏi thẳng \emph{«đoạn này có trả lời được câu hỏi không?»} thì đoạn 3 trả
      lời \no{không}, đoạn 12 cũng \no{không}.
      \vspace{0.4em}

      cả hai đều đúng — và ta \no{loại oan $\{3,12\}$}, tức loại đúng loại câu hỏi
      nhiều chặng mà cả dự án nhắm tới.
    \end{badb}
  \end{column}
  \end{columns}

  \vspace{0.7em}
  \begin{keyb}
    chẻ ra thì mỗi mảnh có \emph{đúng một} đoạn khẳng định nó, ghép lại mới thành
    câu trả lời. đó chính là hình dạng của một câu hỏi nhiều chặng.
  \end{keyb}
\end{frame}

\begin{frame}[fragile]{c0 · ba thứ rơi ra miễn phí từ bảng phủ}
  \framesubtitle{c0}
  \lead{bảng phủ dựng xong rồi thì ba thứ sau có ngay, không tốn thêm gì.}

  \begin{columns}[t]
  \begin{column}{0.46\textwidth}
    \begin{keyb}[title={1 · cửa thoát}]
      mảnh nào \emph{không đoạn nào} nói ra $\Rightarrow$ trong pool
      \no{không tồn tại} tập đủ. bỏ truy vấn ngay, \yes{trước khi tiêu một lượt
      llm nào}.
    \end{keyb}
    \begin{bx}[title={3 · đoán trước cỡ đáp số}]
      $J$ mảnh thì đáp số \emph{nên} có khoảng $J$ đoạn. ví dụ: $J = 2$, và đáp số
      thật đúng là 2 đoạn.
    \end{bx}
  \end{column}
  \begin{column}{0.5\textwidth}
    \begin{bx}[title={2 · danh sách ứng viên, giải chính xác}]
      \begin{dia}
mảnh 1 được phủ bởi:  {3, 7}
mảnh 2 được phủ bởi:  {12, 23, 31}

tập nhỏ nhất phủ hết cả hai mảnh:
  {3,12} {3,23} {3,31}
  {7,12} {7,23} {7,31}
                 -> đúng 6 ứng viên
      \end{dia}
      \vspace{0.25em}
      từ \hl{hơn nghìn tỉ} tập con xuống \hl{6 ứng viên}, bằng một phép tính tổ hợp
      chứ không phải phỏng đoán.
      \vspace{0.25em}

      \dd{bài toán này tên là \emph{set cover}; ở cỡ $J \le 4$ thì giải vét cạn
          trong mili-giây}
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{đo «trả lời đúng» — con số $v(S)$}
  \framesubtitle{công cụ dùng ở c1 c3 c4 c6}
  \lead{$v(S)$ là điểm khi \emph{chỉ} cho model đọc tập $S$ rồi bắt trả lời.
        càng gần 1 càng đúng. mỗi lần đo tốn \hl{hai lượt gọi llm}.}

  \begin{columns}[t]
  \begin{column}{0.52\textwidth}
    \begin{bx}[title={lượt 1 · bắt model trả lời, chỉ từ $S$}]
      \begin{dia}
Answer using ONLY the passages below.
If insufficient, say: INSUFFICIENT.

[1] <nội dung đoạn 3>
[2] <nội dung đoạn 12>
Question: ai đạo diễn phim thắng oscar 1994?
Answer:
      \end{dia}
    \end{bx}
    \begin{bx}[title={lượt 2 · một model khác chấm đúng/sai}]
      \begin{dia}
Gold answer: Robert Zemeckis
Also acceptable: Zemeckis · Robert Lee Zemeckis
Candidate answer: <câu model vừa trả lời>
Correct? One word: Yes or No.
      \end{dia}
    \end{bx}
  \end{column}
  \begin{column}{0.44\textwidth}
    \lead{không lấy chữ «Yes», mà lấy \emph{độ tin} của model vào chữ đó:}
    \[
      v(S) \;=\;
      \frac{\sum_{t \in \mathcal{Y}} e^{\ell_t}}
           {\sum_{t \in \mathcal{Y}} e^{\ell_t} + \sum_{t \in \mathcal{N}} e^{\ell_t}}
    \]
    \dd{$\ell_t$ = điểm model gán cho từ $t$ ở vị trí đầu ·
        $\mathcal{Y}$ = \{Yes, yes, ...\}, $\mathcal{N}$ = \{No, no, ...\}}

    \vspace{0.4em}
    \begin{keyb}[title={vì sao cần một số, không phải đúng/sai}]
      đúng/sai cho $0$ hoặc $1$, nên hàng chục tập con \no{hòa nhau} và không so
      được tập nào hơn tập nào. có số thực thì mới xếp hạng được.
    \end{keyb}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{c1 c3 c4 · ba phép đo mốc, rồi chốt ngưỡng}
  \framesubtitle{c1 · c3 · c4}
  \lead{$v(S) = 0{,}71$ là cao hay thấp? không trả lời được, vì mỗi câu hỏi một
        độ khó. nên phải đo \emph{mốc} của riêng câu hỏi đó trước.}

  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{bx}[title={ba lượt đo, mỗi lượt một mốc}]
      \begin{dia}
c1  v(rỗng) = 0,05   không cho đọc gì
                     -> model biết sẵn bao nhiêu
c3  v(R)    = 0,88   12 đoạn tốt nhất
c4  v(C)    = 0,81   cả 40 đoạn
      \end{dia}
      \vspace{0.25em}
      \hl{trần} $v_{\mathrm{ref}} = \max(\,$ ba số trên, và mọi $v(S)$ đo sau này $)
      = 0{,}88$.
      \vspace{0.25em}

      \dd{$v(C) < v(R)$ \emph{không} phải lỗi: nhồi 40 đoạn làm model phân tán.
          nên trần phải là \textbf{cực đại quan sát được}, không phải $v(C)$}
    \end{bx}
  \end{column}
  \begin{column}{0.47\textwidth}
    \lead{ngưỡng đặt ở $95\%$ quãng từ sàn lên trần:}
    \[
      \tau_q = \theta\,\bigl[\,v_{\mathrm{ref}} - v(\emptyset)\,\bigr] + v(\emptyset),
      \qquad \theta = 0{,}95
    \]
    \vspace{-0.6em}
    \begin{dia}
tau_q = 0,95 (0,88 - 0,05) + 0,05 = 0,84
    \end{dia}
    \vspace{0.2em}
    \begin{keyb}[title={đọc ngưỡng này ra lời}]
      «$S$ đạt được \hl{ít nhất $95\%$} phần mà \emph{cả pool} làm được, tính từ
      mức model tự biết.»
    \end{keyb}
    \begin{bx}
      \dd{\textbf{chốt chặn:} nếu $v_{\mathrm{ref}} - v(\emptyset) < 0{,}05$ thì
          model vốn đã tự trả lời được, truy hồi chẳng đóng góp gì —
          trả $T = \emptyset$ và gắn cờ. ca này có thật, không phải lỗi.}
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{ngưỡng tăng dần mà không tự mâu thuẫn}
  \framesubtitle{c1 · c3 · c4}
  \lead{trần là «cực đại quan sát được», nên đo thêm thì trần có thể tăng, kéo
        ngưỡng tăng theo. vậy tập vừa nâng trần có bị chính ngưỡng mới loại không?}

  \vspace{0.5em}
  \[
    v(S) > v_{\mathrm{ref}}
    \;\Longrightarrow\;
    v_{\mathrm{ref}} := v(S)
    \;\Longrightarrow\;
    \tau_q' = \theta\bigl[v(S) - v(\emptyset)\bigr] + v(\emptyset)
    \;\underbrace{<}_{\text{vì } \theta\,<\,1}\; v(S)
  \]

  \vspace{0.8em}
  \begin{columns}[t]
  \begin{column}{0.48\textwidth}
    \begin{keyb}[title={không, và $\theta < 1$ là lý do duy nhất}]
      ngưỡng mới luôn \yes{thấp hơn} điểm của tập vừa nâng trần. nên tập đó vẫn
      vượt ngưỡng. cập nhật không bao giờ loại chính cái gây ra nó.
    \end{keyb}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{bx}[title={và không phải xét lại quá khứ}]
      $\tau_q$ chỉ \emph{tăng}, nên tập nào đã trượt thì vẫn trượt — không cần
      chấm lại những tập đã bỏ.
      \vspace{0.25em}

      \dd{nếu đặt $\theta = 1$ thì $\tau_q' = v(S)$ đúng bằng điểm của nó, và
          tính chất trên mất}
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}
