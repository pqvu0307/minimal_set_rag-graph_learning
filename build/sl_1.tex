\section{bài toán}

\begin{frame}[fragile]{rag là gì, và nó thiếu gì}
  \lead{hỏi một model ngôn ngữ về sự kiện có thật thì nó hay \emph{bịa}.
        cách chữa phổ biến tên là \hl{rag}: trước khi trả lời, đi tìm vài
        đoạn văn bản liên quan rồi đưa kèm câu hỏi cho model đọc.}

  \begin{dia}
câu hỏi  ->  [ĐI TÌM]  ->  40 đoạn văn bản  ->  [MODEL ĐỌC]  ->  câu trả lời
                                ^
                                |
                       trong 40 đoạn này, đoạn nào THẬT SỰ tạo ra
                       câu trả lời?  hiện không ai biết.
  \end{dia}

  \vspace{0.4em}
  \begin{columns}[t]
  \begin{column}{0.49\textwidth}
    \begin{badb}[title={vì sao «không ai biết» là vấn đề}]
      câu trả lời không kèm nguồn là câu trả lời \no{không dùng được} trong y tế,
      pháp lý, tài chính — người đọc không kiểm lại được.
    \end{badb}
  \end{column}
  \begin{column}{0.49\textwidth}
    \begin{bx}[title={và nó cũng chặn việc cải thiện hệ thống}]
      không biết đoạn nào đúng thì \no{không dạy} được bộ đi tìm lần sau
      tìm tốt hơn — không có nhãn để học.
    \end{bx}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{ví dụ chạy suốt buổi}
  \framesubtitle{mọi bước sau đều quay lại ví dụ này}
\begin{dia}
câu hỏi   ai đạo diễn bộ phim thắng giải oscar phim hay nhất năm 1994?
đáp án    Robert Zemeckis

hệ thống lấy về 40 đoạn.  ba đoạn đáng chú ý:

  đoạn  3   "Forrest Gump thắng oscar phim hay nhất năm 1994."
  đoạn 12   "Robert Zemeckis đạo diễn Forrest Gump."
  đoạn 19   "Forrest Gump là hiện tượng văn hóa của thập niên 1990."

  ... và 37 đoạn khác, phần lớn không liên quan
\end{dia}

  \vspace{0.3em}
  \begin{columns}[t]
  \begin{column}{0.47\textwidth}
    \begin{keyb}[title={đáp số mà tầng a phải tự tìm ra}]
      \hl{đúng hai đoạn: 3 và 12.}
      \vspace{0.25em}

      đoạn 3 cho biết \emph{phim nào}, đoạn 12 cho biết \emph{ai đạo diễn}.
      \no{không đoạn nào một mình} trả lời được — phải \hl{bắc cầu} qua hai đoạn.
    \end{keyb}
  \end{column}
  \begin{column}{0.47\textwidth}
    \begin{badb}[title={đoạn 19 là cái bẫy của cả bài}]
      đoạn 19 \no{không} nói ai đạo diễn, cũng \no{không} nói phim thắng giải.
      \vspace{0.25em}

      nhưng nó \emph{nhắc tên phim} — và chỉ cần thế là model nhớ ra đáp án từ
      kiến thức sẵn có rồi trả lời đúng. ta sẽ phải loại nó.
    \end{badb}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{hai câu hỏi, hai công cụ}
  \lead{«đoạn nào» và «vì sao tìm ra» là hai bài toán khác hẳn nhau, nên dự án
        tách làm hai tầng.}

  \begin{columns}[t]
  \begin{column}{0.48\textwidth}
    \begin{bx}[title={tầng a — đoạn nào gánh câu trả lời}]
      trong 40 đoạn đã lấy về, tập nào \emph{thật sự} đủ để trả lời,
      và không đoạn nào thừa?
      \vspace{0.3em}

      \dd{ở ví dụ trên, đáp số là $\{3, 12\}$}
    \end{bx}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{bx}[title={tầng b — vì sao bộ đi tìm tìm ra chúng}]
      hệ thống đi tìm bằng cách lần theo một \emph{đồ thị tri thức}.
      \emph{đường} nào đã dẫn nó tới đoạn 3 và đoạn 12?
      \vspace{0.3em}

      \dd{ví dụ: câu hỏi $\to$ «Forrest Gump» $\to$ «Zemeckis» $\to$ đoạn 12}
    \end{bx}
  \end{column}
  \end{columns}

  \vspace{0.7em}
  \begin{keyb}
    hai tầng nối nhau bằng \hl{một thứ duy nhất}: tầng a trả về tập đoạn $T$,
    tầng b nhận $T$ rồi đi tìm đường. \no{không điểm số nào đi qua ranh giới} —
    lý do ở cuối buổi.
  \end{keyb}
\end{frame}
